<script>
    // Teachable Machine 모델 URL
    const URL = "https://teachablemachine.withgoogle.com/models/GihtqkmqM/";

    let model, webcam, maxPredictions;
    let isRunning = false;
    let lastSpoken = ""; // 중복 음성 안내 방지용

    function speak(text) {
        // 동일한 안내가 계속 반복되는 것을 방지 (약간의 쿨타임)
        if (lastSpoken === text && window.speechSynthesis.speaking) return;
        
        window.speechSynthesis.cancel();
        const msg = new SpeechSynthesisUtterance(text);
        msg.lang = 'ko-KR';
        msg.rate = 1.0;
        lastSpoken = text;
        window.speechSynthesis.speak(msg);
    }

    async function init() {
        const btn = document.getElementById("start-btn");
        const loader = document.getElementById("loader");
        const errorMsg = document.getElementById("error-message");

        btn.disabled = true;
        btn.innerText = "모델 로딩 중...";
        loader.style.display = "block";
        errorMsg.style.display = "none";

        try {
            const modelURL = URL + "model.json";
            const metadataURL = URL + "metadata.json";

            model = await tmImage.load(modelURL, metadataURL);
            maxPredictions = model.getTotalClasses();

            const flip = true; 
            webcam = new tmImage.Webcam(400, 400, flip);
            await webcam.setup(); 
            await webcam.play();
            
            loader.style.display = "none";
            btn.innerText = "실시간 감지 중";
            
            document.getElementById("webcam-container").appendChild(webcam.canvas);
            
            isRunning = true;
            window.requestAnimationFrame(loop);
        } catch (e) {
            console.error(e);
            loader.style.display = "none";
            btn.disabled = false;
            btn.innerText = "다시 시도";
            errorMsg.innerText = "카메라 권한을 허용했는지 확인해주세요.";
            errorMsg.style.display = "block";
        }
    }

    async function loop() {
        if (!isRunning) return;
        webcam.update();
        await predict();
        window.requestAnimationFrame(loop);
    }

    async function predict() {
        const prediction = await model.predict(webcam.canvas);
        const statusDiv = document.getElementById("status-display");
        const maskVal = document.getElementById("val-mask");
        const noMaskVal = document.getElementById("val-nomask");

        let topClass = "";
        let topProb = 0;

        for (let i = 0; i < maxPredictions; i++) {
            const className = prediction[i].className;
            const probability = prediction[i].probability;
            const percent = (probability * 100).toFixed(0) + "%";

            if (className === "마스크") maskVal.innerText = percent;
            if (className === "노마스크") noMaskVal.innerText = percent;

            if (probability > topProb) {
                topProb = probability;
                topClass = className;
            }
        }

        // 결과 업데이트 (신뢰도 85% 이상일 때만 상태 변경)
        if (topProb > 0.85) {
            if (topClass === "마스크") {
                statusDiv.innerHTML = "<span>✅ 환영합니다</span>";
                statusDiv.className = "status-badge status-mask";
                if (!window.speechSynthesis.speaking || lastSpoken !== "환영합니다") {
                    speak("환영합니다");
                }
            } else if (topClass === "노마스크") {
                statusDiv.innerHTML = "<span>❌ 마스크를 써주세요</span>";
                statusDiv.className = "status-badge status-nomask";
                if (!window.speechSynthesis.speaking || lastSpoken !== "마스크를 쓰고 입장해주세요") {
                    speak("마스크를 쓰고 입장해주세요");
                }
            }
        } else {
            statusDiv.innerHTML = "<span>인식 중...</span>";
            statusDiv.className = "status-badge status-wait";
        }
    }
</script>
